-- Hàm đọc nội dung file
local function readFileRaw(filename)
    -- Biến lưu kết quả
    local ok, d = pcall(function()
        -- Kiểm tra xem hàm readfile có tồn tại không
        if readfile and isfile and isfile(filename) then
            -- Nếu có thì đọc file và trả về nội dung
            return readfile(filename)
        end
        -- Nếu không có file thì trả về chuỗi rỗng
        return ""
    end)
    -- Nếu đọc thành công thì trả về dữ liệu, ngược lại trả về chuỗi rỗng
    return ok and d or ""
end

-- Hàm ghi nội dung vào file
local function writeFileRaw(filename, content)
    -- Bọc trong pcall để tránh lỗi
    pcall(function()
        -- Kiểm tra xem hàm writefile có tồn tại không
        if writefile then
            -- Ghi nội dung vào file
            writefile(filename, content)
        end
    end)
end

end

-- Hàm kiểm tra định dạng key có hợp lệ không
local function validateKey(k)
    -- Kiểm tra kiểu dữ liệu phải là string
    if type(k) ~= "string" then
        return false
    end
    -- Kiểm tra độ dài key phải lớn hơn 4 ký tự
    if #k < 4 then
        return false
    end
    -- Biến đếm số chữ in hoa
    local upperCount = 0
    -- Duyệt qua từng ký tự trong key
    for i = 1, #k do
        -- Lấy ký tự tại vị trí i
        local char = string.sub(k, i, i)
        -- Nếu là chữ in hoa thì tăng biến đếm
        if char:match("%u") then
            upperCount = upperCount + 1
        end
    end
    -- Yêu cầu phải có đúng 3 chữ in hoa
    if upperCount ~= 3 then
        return false
    end
    -- Yêu cầu key phải kết thúc bằng dấu chấm và khoảng trắng
    if not string.match(k, "%.%s+$") then
        return false
    end
    -- Nếu thỏa mãn tất cả thì trả về true
    return true
end

-- Hàm kiểm tra key hiện tại có còn hạn không
local function keyValidNow()
    -- Đọc nội dung file key
    local raw = readFileRaw(KEY_FILE)
    -- Nếu file rỗng thì trả về false
    if raw == "" then
        return false
    end
    -- Tìm dấu phân cách giữa key và thời gian
    local a, b = string.find(raw, "|", 1, true)
    -- Nếu không tìm thấy thì trả về false
    if not a then
        return false
    end
    -- Tách lấy phần key
    local k  = string.sub(raw, 1, a - 1)
    -- Tách lấy phần thời gian (timestamp)
    local ts = tonumber(string.sub(raw, b + 1))
    -- Kiểm tra key và thời gian có hợp lệ không
    if not k or not ts or not validateKey(k) then
        return false
    end
    -- Kiểm tra thời gian hiện tại so với thời gian lưu (3600 giây = 1 tiếng)
    if (os.time() - ts) >= 3600 then
        return false
    end
    -- Nếu còn hạn thì trả về true
    return true
end

-- Hàm kiểm tra key này đã từng được sử dụng chưa
local function isKeyUsed(k)
    -- Đọc danh sách các key đã dùng
    local used = readFileRaw(USED_KEYS_FILE)
    -- Duyệt qua từng dòng trong file
    for line in string.gmatch(used, "[^\n]+") do
        -- Nếu tìm thấy key trùng khớp
        if line == k then
            return true
        end
    end
    -- Nếu không tìm thấy thì trả về false
    return false
end

-- Hàm đánh dấu key đã được sử dụng (burn key)
local function markKeyUsed(k)
    -- Nếu key đã được dùng rồi thì không làm gì cả
    if isKeyUsed(k) then
        return
    end
    -- Đọc danh sách key đã dùng
    local used = readFileRaw(USED_KEYS_FILE)
    -- Ghi thêm key mới vào danh sách (xuống dòng)
    writeFileRaw(USED_KEYS_FILE, used .. k .. "\n")
end

-- Ký tự ẩn ZWSP (Zero Width Space)
local ZW0 = "\u{200B}"

-- Ký tự ẩn ZWNJ (Zero Width Non-Joiner)
local ZW1 = "\u{200C}"

-- Chữ ký nhận diện tin nhắn của script
local OBS_SIG = "FXADMIN"

-- Hàm mã hóa lệnh thành chuỗi ký tự ẩn
local function encodeObs(cmd, data)
    -- Tạo chuỗi gốc cần mã hóa
    local s = OBS_SIG .. "|" .. cmd .. "|" .. tostring(data)
    -- Bảng chứa các ký tự ẩn
    local out = {}
    -- Duyệt qua từng ký tự trong chuỗi gốc
    for i = 1, #s do
        -- Lấy mã byte của ký tự
        local b = string.byte(s, i)
        -- Duyệt từ bit cao đến bit thấp
        for j = 7, 0, -1 do
            -- Tính giá trị bit hiện tại
            local bit = math.floor(b / 2^j) % 2
            -- Nếu bit = 1 thì thêm ZW1, ngược lại thêm ZW0
            if bit == 1 then
                out[#out+1] = ZW1
            else
                out[#out+1] = ZW0
            end
        end
    end
    -- Nối tất cả ký tự ẩn lại thành 1 chuỗi
    return table.concat(out)
end

-- Hàm giải mã chuỗi ký tự ẩn
local function decodeObs(text)
    -- Nếu chuỗi quá ngắn thì bỏ qua
    if #text < 64 then
        return nil
    end
    -- Bảng chứa các bit
    local bits = {}
    -- Duyệt qua từng ký tự trong chuỗi
    for i = 1, #text do
        -- Lấy ký tự hiện tại
        local c = string.sub(text, i, i)
        -- Nếu là ZW0 thì thêm bit 0
        if c == ZW0 then
            bits[#bits+1] = "0"
        -- Nếu là ZW1 thì thêm bit 1
        elseif c == ZW1 then
            bits[#bits+1] = "1"
        end
    end
    -- Nếu không đủ bit thì bỏ qua
    if #bits < 8 then
        return nil
    end
    -- Bảng chứa các byte giải mã
    local out = {}
    -- Duyệt theo từng nhóm 8 bit
    for i = 1, #bits - 7, 8 do
        -- Biến lưu giá trị byte
        local b = 0
        -- Tính giá trị byte từ 8 bit
        for j = 0, 7 do
            b = b * 2 + tonumber(bits[i+j])
        end
        -- Chuyển byte thành ký tự và thêm vào bảng
        out[#out+1] = string.char(b)
    end
    -- Nối tất cả ký tự lại thành chuỗi
    local s = table.concat(out)
    -- Kiểm tra chữ ký nhận diện
    if string.sub(s, 1, #OBS_SIG) ~= OBS_SIG then
        return nil
    end
    -- Tách các phần bằng dấu |
    local parts = string.split(s, "|")
    -- Nếu không đủ 3 phần thì bỏ qua
    if #parts < 3 then
        return nil
    end
    -- Trả về lệnh và dữ liệu
    return parts[2], parts[3]
end

-- Hàm gửi tin nhắn chat (dùng để gửi ký tự ẩn hoặc thông báo)
local function ChatSpam(msg)
    -- Bọc trong pcall để tránh lỗi
    pcall(function()
        -- Lấy dịch vụ TextChatService
        local tcs = game:GetService("TextChatService")
        -- Nếu dùng TextChatService mới
        if tcs and tcs.ChatVersion == Enum.ChatVersion.TextChatService then
            -- Lấy kênh chat
            local ch = tcs:FindFirstChild("TextChannels")
            -- Nếu tìm thấy kênh RBXGeneral
            if ch and ch:FindFirstChild("RBXGeneral") then
                -- Gửi tin nhắn
                ch.RBXGeneral:SendAsync(msg)
            end
        -- Nếu dùng chat cũ
        else
            -- Lấy sự kiện chat cũ
            local ev = ReplicatedStorage:FindFirstChild("DefaultSystemChatEvents")
            -- Nếu tìm thấy
            if ev then
                -- Lấy sự kiện gửi tin nhắn
                local say = ev:FindFirstChild("SayMessageRequest")
                -- Nếu tìm thấy
                if say then
                    -- Gửi tin nhắn
                    say:FireServer(msg, "All")
                end
            end
        end
    end)
end

-- Hàm lắng nghe animation M2 của đối thủ
local function setupM2DynamicListener(p)
    -- Nếu là bản thân thì bỏ qua
    if p == LocalPlayer then return end
    
    -- Hàm xử lý khi nhân vật đối thủ xuất hiện
    local function onChar(char)
        if not char then return end
        
        -- Lấy Humanoid và Root
        local hum = char:WaitForChild("Humanoid", 5)
        local root = char:WaitForChild("HumanoidRootPart", 5)
        if not hum or not root then return end
        
        -- Lắng nghe sự kiện chơi animation
        hum.AnimationPlayed:Connect(function(track)
            -- Nếu Auto Block đang tắt thì bỏ qua
            if not isBlockActive then return end
            
            -- Lấy ID animation
            local animId = track.Animation and track.Animation.AnimationId
            
            -- Nếu đúng là animation M2 (đòn mạnh)
            if animId == M2_TargetID then
                -- Lấy nhân vật mình
                local myChar = LocalPlayer.Character
                if myChar and myChar:FindFirstChild("HumanoidRootPart") then
                    -- Tính khoảng cách
                    local myPos = myChar.HumanoidRootPart.Position
                    local cPos = root.Position
                    local dx, dy, dz = cPos.X - myPos.X, cPos.Y - myPos.Y, cPos.Z - myPos.Z
                    
                    -- Nếu trong tầm đánh
                    if (dx * dx + dy * dy + dz * dz) <= COMBAT_RANGE_SQ then
                        -- Tăng biến đếm M2
                        activeM2Count = activeM2Count + 1
                        
                        -- Thả block ra (để ăn đòn M2)
                        rawUnblock()
                        
                        -- Biến kiểm tra đã dọn dẹp chưa
                        local cleaned = false
                        local stoppedConn, endedConn, diedConn
                        local timeoutTask
                        
                        -- Hàm dọn dẹp
                        local function cleanup()
                            if cleaned then return end
                            cleaned = true
                            activeM2Count = math.max(0, activeM2Count - 1)
                            if stoppedConn then stoppedConn:Disconnect() end
                            if endedConn then endedConn:Disconnect() end
                            if diedConn then diedConn:Disconnect() end
                            if timeoutTask then task.cancel(timeoutTask) end
                        end
                        
                        -- Gắn sự kiện kết thúc animation
                        stoppedConn = track.Stopped:Connect(cleanup)
                        endedConn = track.Ended:Connect(cleanup)
                        diedConn = hum.Died:Connect(cleanup)
                        
                        -- Hẹn giờ tự động dọn dẹp sau khi animation kết thúc
                        local animDuration = (track.Length > 0 and track.Length) or 1.5
                        timeoutTask = task.delay(animDuration + 0.1, cleanup)
                    end
                end
            end
        end)
    end
    
    -- Gắn sự kiện cho nhân vật hiện tại
    if p.Character then onChar(p.Character) end
    
    -- Gắn sự kiện khi đối thủ hồi sinh
    p.CharacterAdded:Connect(onChar)
end

        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            drag = true
            local rx = input.Position.X - tr.AbsolutePosition.X
            setV(minV + math.clamp(rx / tr.AbsoluteSize.X, 0, 1) * (maxV - minV))
        end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if drag and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            local rx = input.Position.X - tr.AbsolutePosition.X
            setV(minV + math.clamp(rx / tr.AbsoluteSize.X, 0, 1) * (maxV - minV))
        end
    end)
    UserInputService.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then 
            drag = false 
        end
    end)
end

-- Hàm kiểm tra người chơi có hợp lệ để tấn công không
local function Valid(p)
    if not p or p == LocalPlayer or Allies[p.Name] then 
        return false 
    end
    local c = p.Character
    if not c then 
        return false 
    end
    local r = c:FindFirstChild("HumanoidRootPart")
    local h = c:FindFirstChildOfClass("Humanoid")
    return r and h and h.Health > 0
end

-- Hàm tìm người chơi gần nhất
local function FindNearest(range)
    local mr = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
    if not mr then return nil end
    local best, bd = nil, range or 100
    for _, p in ipairs(Players:GetPlayers()) do
        if Valid(p) then
            local r = p.Character:FindFirstChild("HumanoidRootPart")
            if r then
                local d = (mr.Position - r.Position).Magnitude
                if d < bd then 
                    bd = d
                    best = p 
                end
            end
        end
    end
    return best
end

-- Hàm tìm người chơi theo tên
local function FindByName(name)
    if not name or name == "" then return nil end
    name = string.lower(name)
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= LocalPlayer and string.find(string.lower(p.Name), name, 1, true) then 
            return p 
        end
    end
    return nil
end

-- Hàm lấy vũ khí (Tool) đang cầm
local function GetTool()
    local c = LocalPlayer.Character
    if not c then return nil end
    local t = c:FindFirstChildOfClass("Tool")
    if t then return t end
    local bp = LocalPlayer:FindFirstChild("Backpack")
    if bp then
        for _, x in ipairs(bp:GetChildren()) do
            if x:IsA("Tool") then 
                x.Parent = c
                return x 
            end
        end
    end
    return nil
end

-- Hàm tấn công nhanh (dùng cho chế độ EZY)
local function EzyFastAttack(target)
    local c = LocalPlayer.Character
    if not c then return end
    local hum = c:FindFirstChildOfClass("Humanoid")
    if not hum or hum.Health <= 0 then return end
    
    -- Dừng các animation đánh cũ
    for _, track in ipairs(hum:GetPlayingAnimationTracks()) do
        pcall(function()
            if track.Priority == Enum.AnimationPriority.Action or track.Priority == Enum.AnimationPriority.Action2 then
                track:Stop(0.1)
            end
        end)
    end

    -- Kích hoạt vũ khí
    local tool = c:FindFirstChildOfClass("Tool")
    if tool then pcall(function() tool:Activate() end) end

    -- Gửi Remote
    if clientSideHitbox and target then
        pcall(function() clientSideHitbox:FireServer(target, buffer.fromstring("10069.74")) end)
    end
    if combatRemote then
        pcall(function() combatRemote:FireServer("M1") end)
        pcall(function() combatRemote:FireServer("Attack") end)
    end
    for _, r in ipairs(Remotes) do
        pcall(function() r:FireServer("M1") end)
        pcall(function() r:FireServer() end)
    end
    for _, r in ipairs(ManualRemotes) do
        pcall(function() r:FireServer("M1") end)
        pcall(function() r:FireServer("Attack") end)
        pcall(function() r:FireServer() end)
    end
end

-- Hàm tấn công tất cả (dùng cho M1 Spam, Spam Đánh)
local function HitAll(target)
    local c = LocalPlayer.Character
    if not c then return end
    
    -- Kích hoạt tất cả vũ khí
    for _, t in ipairs(c:GetChildren()) do
        if t:IsA("Tool") then pcall(function() t.Enabled = true; t:Activate() end) end
    end
    
    -- Gửi Remote tự động
    for _, r in ipairs(Remotes) do
        pcall(function() r:FireServer("M1") end)
        pcall(function() r:FireServer() end)
    end
    
    -- Gửi Remote thủ công
    for _, r in ipairs(ManualRemotes) do
        pcall(function() r:FireServer("M1") end)
        pcall(function() r:FireServer("Attack") end)
        pcall(function() r:FireServer() end)
    end
    
    -- Gửi Hitbox
    if clientSideHitbox and target then
        pcall(function() clientSideHitbox:FireServer(target, buffer.fromstring("10069.74")) end)
    end
    
    if combatRemote then
        pcall(function() combatRemote:FireServer("M1") end)
        pcall(function() combatRemote:FireServer("Attack") end)
    end
end
